% Beamer sprint-review template — agile sprint review/demo (~10-15 min, 8-12 frames).
% Shape: goal · shipped · demo · metrics · slipped · retro · next sprint · asks.
% Compile: latexmk -pdf -interaction=nonstopmode -halt-on-error main.tex
\documentclass[aspectratio=169,12pt]{beamer}

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[english]{babel}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{hyperref}

\usetheme{Minimal}                     % requires beamerthemeMinimal.sty on TEXINPUTS
\setbeamertemplate{navigation symbols}{}

\title[Sprint <N> Review]{Sprint <N> Review}
\subtitle{<DATE-RANGE>}
\author[<SHORT-NAME>]{<NAME>}
\institute[<TEAM>]{<TEAM> $\cdot$ <PRODUCT-AREA>}
\date{\today}

\begin{document}

\begin{frame}[plain]
  \titlepage
\end{frame}

\begin{frame}{Sprint goal}
  \begin{quote}
    <One-sentence goal as committed at sprint start>
  \end{quote}
  \textbf{Met?} \checkmark{} / partial / not met --- and one sentence why.
\end{frame}

\begin{frame}{What we shipped}

    \item \textbf{<FEATURE / FIX 2>} --- one-line value statement
    \item \textbf{<FEATURE / FIX 3>} --- one-line value statement
  \end{itemize}
\end{frame}

\begin{frame}{What slipped \& why}
  \begin{itemize}
    \item \textbf{<COMMITTED ITEM>} --- slipped because: <root cause>. Plan: <next sprint or descope>.

\end{frame}

\begin{frame}{Retro headlines}
  \begin{itemize}

\end{frame}

\begin{frame}{Next sprint commitment}
