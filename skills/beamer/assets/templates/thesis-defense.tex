\begin{frame}{Why this thesis}
  \begin{itemize}
    \item Service meshes are now standard infrastructure
    \item Two paradigms compete: \textbf{sidecar} (per-pod proxy) vs.\ \textbf{ambient} (shared per-node)
    \item Their telemetry is \emph{assumed comparable} --- but is it, under canary deployments with fault injection?
  \end{itemize}
  \note{90s. Set the stakes: automated canary decisions ride on this telemetry.}
\end{frame}

\begin{frame}{Research questions}
  \begin{itemize}
    \item \textbf{RQ1 --- Decision correctness:} how do false-positive and false-negative rates of automated canary abort decisions differ between modes?
    \item \textbf{RQ2 --- Remediation speed:} how does end-to-end remediation latency compare?
    \item \textbf{RQ3 --- Telemetry fidelity:} what differences in metric completeness, label attribution, and signal stability exist?
  \end{itemize}
\end{frame}

\begin{frame}{Methodology}
  \begin{itemize}
    \item Single independent variable: mesh mode
    \item Clean-state reset between every run (GitOps-controlled mode switch)
    \item 10 runs per mode at each fault percentage
    \item Timing chain: $T_0$ (fault start) $\to$ $T_1$ (alert fires) $\to$ $T_5$ (traffic shift complete)
  \end{itemize}
\end{frame}
